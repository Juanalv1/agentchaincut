#!/usr/bin/env python3
"""AgentChainCut PreToolUse hook (generated). Decides MCP tool calls with the computed policy."""
import json
import os
import sys

from acc.policy_engine import check

POLICY = json.loads('{"agents": [{"name": "support-agent", "tools": {"billing.invoice_list": {"constraints": [{"arg": "customer_id", "ctx": "customer.id", "invariant": "inv-1", "op": "eq"}], "mode": "scoped"}, "billing.invoice_list_all": {"mode": "deny"}, "billing.invoice_read": {"constraints": [{"arg": "invoice_id", "ctx": "customer.invoice_ids", "invariant": "inv-2", "op": "in"}], "mode": "scoped"}, "billing.refund": {"constraints": [{"arg": "invoice_id", "ctx": "customer.invoice_ids", "invariant": "inv-3", "op": "in"}], "mode": "approval"}, "mail.send": {"constraints": [{"arg": "to", "ctx": "ticket.customer.email", "invariant": "inv-4", "op": "eq"}], "mode": "scoped"}, "tickets.add_note": {"constraints": [{"arg": "ticket_id", "ctx": "ticket.id", "invariant": "inv-5", "op": "eq"}], "mode": "scoped"}, "tickets.get": {"constraints": [{"arg": "ticket_id", "ctx": "ticket.id", "invariant": "inv-6", "op": "eq"}], "mode": "scoped"}}}], "conflicts": [{"core": ["billing.refund"], "fixes": [{"tool": "billing.refund", "type": "approval_edge"}, {"partition": [["tickets.get", "tickets.add_note", "mail.send"], ["billing.invoice_read", "billing.invoice_list", "billing.refund"]], "type": "split_agent"}], "legs": ["U", "S", "X"], "resolved_by": "approval_edge", "task": "refund_request"}], "legs_per_session": {"S": true, "U": true, "X": "bounded"}, "solver": {"broken_tasks": [], "method": "cp-sat", "objective": 8, "status": "OPTIMAL"}}')
PROXY_SERVER = "acc"
SERVERS = {name.split(".", 1)[0] for agent in POLICY["agents"] for name in agent["tools"]}
DECISIONS = {"allow": "allow", "deny": "deny", "approval": "ask"}


def policy_tool(tool_name):
    """mcp__<server>__<tool> or mcp__acc__<server>__<tool> -> <server>.<tool>; None if not ours."""
    if not tool_name.startswith("mcp__"):
        return None
    parts = tool_name[len("mcp__"):].split("__")
    if parts and parts[0] == PROXY_SERVER:
        parts = parts[1:]
    if len(parts) != 2 or parts[0] not in SERVERS:
        return None
    return parts[0] + "." + parts[1]


def load_context():
    path = os.environ.get("ACC_SESSION_CONTEXT")
    if not path:
        return {}
    try:
        with open(path, encoding="utf-8") as fh:
            return json.load(fh)
    except (OSError, ValueError):
        return {}


def main():
    payload = json.load(sys.stdin)
    tool = policy_tool(payload.get("tool_name", ""))
    if tool is None:
        return 0  # not governed by this policy: let Claude Code decide
    decision = check(POLICY, tool, payload.get("tool_input") or {}, load_context())
    print(json.dumps({"hookSpecificOutput": {
        "hookEventName": "PreToolUse",
        "permissionDecision": DECISIONS[decision.decision],
        "permissionDecisionReason": "AgentChainCut " + str(decision.rule) + ": " + decision.reason,
    }}))
    return 0


if __name__ == "__main__":
    sys.exit(main())
